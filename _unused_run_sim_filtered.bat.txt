@echo off
REM run_sim_filtered.bat
REM
REM Runs unified_tracking.py exactly like normal, but filters the
REM "sequence size exceeds remaining buffer" spam out of what you actually
REM see. Three different attempts at suppressing this via carb's log-level
REM settings (channel-level Error, channel-level Fatal, and the top-level
REM /log/outputStreamLevel constructor setting) all failed to touch it -
REM that's a strong sign this specific message is printed directly by a
REM native plugin (fprintf/printf) rather than through carb's logging
REM macros, which means NO log-level setting can ever suppress it from the
REM source. This just strips matching lines from the console output
REM instead, which works regardless of where the message actually comes
REM from internally.
REM
REM Run this INSTEAD of calling python.bat directly:
REM     C:\isaacsim\projects\surveillance-proj\run_sim_filtered.bat

C:\isaacsim\python.bat C:\isaacsim\projects\surveillance-proj\unified_tracking.py 2^>^&1 | findstr /v /c:"sequence size exceeds remaining buffer"
